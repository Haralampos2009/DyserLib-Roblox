local function mksfx(id, vol, pitch)
    local ok, s = pcall(function()
        local snd = Instance.new("Sound")
        snd.SoundId = "rbxassetid://"..tostring(id)
        snd.Volume = vol or 3.0 
        snd.PlaybackSpeed = pitch or 1
        snd.SoundGroup = uiSoundGroup 
        snd.Parent = SoundService
        return snd
    end)
    return ok and s or nil
end

local textCaches = {

local dynamicTextElements = {}

local function trackText(category, obj)
    if obj then table.insert(textCaches[category] or textCaches.Labels, obj) end
end

local function trackColorText(o)
    if o then table.insert(dynamicTextElements, o) end
end

local function updateTextSize(category, size)
    if not textCaches[category] then return end
    for _, obj in ipairs(textCaches[category]) do
        if obj and obj.Parent then
            obj.TextSize = size
        end
    end
end

local function setTextColor(color)
    COLORS.text = color
    for _, o in ipairs(dynamicTextElements) do
        if o and o.Parent then
            local isSelected = false
            
            for _, btnData in ipairs(Nav.topTabBtns) do
                if btnData.label == o and btnData.button == Nav.selectedTop then isSelected = true end
            end
            for _, btnData in ipairs(Nav.rightTabBtns) do
                if btnData.label == o and btnData.button == Nav.selectedRight then isSelected = true end
            end
            if o.Name == "SideTabText" then
                for _, btnData in ipairs(Nav.sideButtons) do
                    if btnData.button == o.Parent and btnData.button == Nav.selectedSide then isSelected = true end
                end
            end
            
            if not isSelected then
                o.TextColor3 = color
            end
        end
    end
end

-- ===================== UTILITIES =====================
local function scaledC(c,s) return Color3.new(math.clamp(c.R*s,0,1),math.clamp(c.G*s,0,1),math.clamp(c.B*s,0,1)) end
local function corner(p,r) local c=Instance.new("UICorner"); c.CornerRadius=UDim.new(0,r); c.Parent=p; return c end
local function stroke(p,col,th,tr)
    local s=Instance.new("UIStroke"); s.Color=col; s.Thickness=th or 1
    s.Transparency=tr or 0; s.ApplyStrokeMode=Enum.ApplyStrokeMode.Border; s.Parent=p; return s
end
local function gradient(p,a,b,rot)
    local g=Instance.new("UIGradient"); g.Color=ColorSequence.new(a,b); g.Rotation=rot or 90; g.Parent=p; return g
end
local function label(p, cls, props)
    local category = props.Category
    props.Category = nil
    
    local o=Instance.new(cls)
    o.BackgroundTransparency=props.BackgroundTransparency or 1
    o.Text=props.Text or ""; 
    local isStandardColor = (props.TextColor3 == nil or props.TextColor3 == COLORS.text)
    o.TextColor3=props.TextColor3 or COLORS.text
    
    o.TextSize=props.TextSize or 14; o.Font=props.Font or Enum.Font.Code
    o.TextXAlignment=props.TextXAlignment or Enum.TextXAlignment.Left
    o.TextYAlignment=props.TextYAlignment or Enum.TextYAlignment.Center
    o.BorderSizePixel=0
    
    if cls == "TextLabel" or cls == "TextButton" or cls == "TextBox" then
        o.TextScaled = false
    end

    for k,v in pairs(props) do pcall(function() o[k]=v end) end
    if p then o.Parent=p end
    
    if isStandardColor then
        trackColorText(o)
    end
    trackText(category or "Labels", o)
    
    return o
end
local function pad(p,l,r,t,b)
    local u=Instance.new("UIPadding",p)
    if l then u.PaddingLeft=UDim.new(0,l) end; if r then u.PaddingRight=UDim.new(0,r) end
    if t then u.PaddingTop=UDim.new(0,t) end; if b then u.PaddingBottom=UDim.new(0,b) end
end

-- ===================== BUBBLES =====================
local bSeed=24680
local function bubble(parent,count,noCenter,getActive,customColor,onAccent)
    parent.ClipsDescendants=true
    local function rnd() bSeed=(bSeed*1103515245+12345)%2147483648; return bSeed/2147483648 end
    
    local ic = onAccent and Color3.fromRGB(255,255,255)
              or customColor and scaledC(customColor,0.55)
              or scaledC(COLORS.accent,1.7)
    local transp = onAccent and .55 or customColor and .35 or .55
    local strokeTranp = onAccent and .7 or customColor and .45 or .25
    
    for _=1,(count or 8) do
        local d=noCenter and (3+math.floor(rnd()*4)) or (4+math.floor(rnd()*7))
        local py=noCenter and (rnd()<.5 and .04+rnd()*.2 or .76+rnd()*.2) or rnd()
        local b=Instance.new("Frame",parent)
        b.AnchorPoint=Vector2.new(.5,.5); b.Position=UDim2.new(rnd(),0,py,0)
        b.Size=UDim2.fromOffset(d,d); b.BackgroundColor3=ic; b.BackgroundTransparency=transp
        b.BorderSizePixel=0; b.ZIndex=6; b.Active=false
        b.Visible=customColor and true or (isUIBubbles and (getActive==nil or getActive()))
        corner(b,50)
        local bs=Instance.new("UIStroke",b); bs.Color=ic; bs.Thickness=1; bs.Transparency=strokeTranp
        
        if not customColor and not onAccent then
            table.insert(Acc.texParts,{f=b,s=bs})
            table.insert(Acc.bubbles,{f=b,ga=getActive})
        elseif not customColor then
            table.insert(Acc.bubbles,{f=b,ga=getActive})
        end
    end
end
local function setBubbles(v)
    isUIBubbles=v
    for _,item in ipairs(Acc.bubbles) do
        if item.f and item.f.Parent then item.f.Visible=v and (item.ga==nil or item.ga()) end
    end
end

-- ===================== NAVIGATION =====================
local function showPage(page)
    for _,e in ipairs(Nav.sideButtons) do e.page.Visible=false end
    if page then page.Visible=true end
end

local function selectSide(btn,grad,name,page)
    for _,e in ipairs(Nav.sideButtons) do
        e.gradient.Enabled=true
        e.button.BackgroundColor3=Color3.fromRGB(17,17,18)
        if e.fillGrad then e.fillGrad.Enabled=false end
        if e.activeStroke then e.activeStroke.Transparency=1 end
        if e.inactiveStroke then e.inactiveStroke.Transparency=.2 end
        local lbl = e.button:FindFirstChild("SideTabText")
        if lbl then lbl.TextColor3=COLORS.text end
    end
    Nav.selectedSide=btn
    grad.Enabled=false
    btn.BackgroundColor3=COLORS.accent
    for _,e in ipairs(Nav.sideButtons) do
        if e.button==btn then
            if e.fillGrad then e.fillGrad.Enabled=true end
            if e.activeStroke then e.activeStroke.Transparency=0 end
            if e.inactiveStroke then e.inactiveStroke.Transparency=1 end
            break
        end
    end
    local actLbl = btn:FindFirstChild("SideTabText")
    if actLbl then actLbl.TextColor3=COLORS.white end
    showPage(page)
end

local function selectTop(btn,name)
    for _,e in ipairs(Nav.topTabBtns) do
        e.button.BackgroundColor3=Color3.fromRGB(8,8,9)
        if e.label then e.label.TextColor3=COLORS.text end
    end
    Nav.selectedTop=btn
    btn.BackgroundColor3=COLORS.accent
    for _,e in ipairs(Nav.topTabBtns) do
        if e.button==btn and e.label then e.label.TextColor3=COLORS.white end
    end
    -- скрываем все API-панели (они принадлежат вкладкам добавленным через AddTopTab)
    if apiSidePanels then
        for _, data in pairs(apiSidePanels) do
            if data.panel and data.panel.Parent then data.panel.Visible = false end
            if data.leftPane and data.leftPane.Parent then data.leftPane.Visible = false end
        end
    end
    -- показываем системную боковую панель и leftPane
    if sidePanel and sidePanel.Parent then sidePanel.Visible = true end
    if leftPane and leftPane.Parent then leftPane.Visible = true end
    local allowed=Nav.topToSide[name] or {}
    local allowSet={}; for _,n in ipairs(allowed) do allowSet[n]=true end
    for _,e in ipairs(Nav.sideButtons) do
        e.button.Visible=allowSet[e.name]==true
    end
    local first=nil
    for _,n in ipairs(allowed) do
        for _,e in ipairs(Nav.sideButtons) do
            if e.name==n then first=e; break end
        end
        if first then break end
    end
    if first then selectSide(first.button,first.gradient,first.name,first.page)
    else showPage(nil) end

    if Nav.sideBottomBtn then
        Nav.sideBottomBtn.Visible = (name == "Main")
    end
end

local function selectRight(btn,name,page)
    for _,e in ipairs(Nav.rightTabBtns) do
        e.button.BackgroundColor3=Color3.fromRGB(8,8,9)
        if e.label then e.label.TextColor3=COLORS.text end
    end
    Nav.selectedRight=btn
    btn.BackgroundColor3=COLORS.accent
    for _,e in ipairs(Nav.rightTabBtns) do
        if e.button==btn and e.label then e.label.TextColor3=COLORS.white end
    end
    for _,p in pairs(Nav.rightTabPgs) do p.Visible=false end
    if page then page.Visible=true end
end

-- reference resolution the menu was designed for; scale grows/shrinks

local REF_SCREEN = Vector2.new(1920, 1080)

-- ===================== DEVICE DETECTION =====================
-- Works out what kind of device the menu is running on, so the menu can fit itself to it:
--   Console = TV / gamepad interface      Phone / Tablet = touch without a keyboard (told apart by screen size)
--   PC      = everything else (keyboard + mouse, also a PC plugged into a TV)
-- NOTE: Roblox only reports the screen size in PIXELS, never its physical size. A PC connected to a 65" 4K TV looks exactly
-- like a PC with a 4K monitor, so that case cannot be detected: pick the size yourself in Settings > Menu Size.
Nav.profileScale = { PC = 1, Tablet = 1, Phone = 1, Console = 1.3 }   -- extra scale per device kind (a console is viewed from a sofa)
Nav.detectDevice = function(viewport)
    local UIS = UserInputService
    local tenFoot = false
    pcall(function() tenFoot = game:GetService("GuiService"):IsTenFootInterface() end)
    local kind
    if tenFoot then
        kind = "Console"
    elseif UIS.TouchEnabled and not UIS.KeyboardEnabled then
        kind = (math.min(viewport.X, viewport.Y) < 600) and "Phone" or "Tablet"
    else
        kind = "PC"
    end
    local inputs = {}
    if UIS.KeyboardEnabled then table.insert(inputs, UIS.MouseEnabled and "Keyboard + Mouse" or "Keyboard") end
    if UIS.TouchEnabled then table.insert(inputs, "Touch") end
    if UIS.GamepadEnabled then table.insert(inputs, "Gamepad") end
    return { kind = kind, input = (#inputs > 0) and table.concat(inputs, " / ") or "Unknown" }
end

local function autoScaleUI()
    local screenSize = screenGui.AbsoluteSize
    if screenSize.X == 0 or screenSize.Y == 0 then return end
    Nav.device = Nav.detectDevice(screenSize)   -- re-checked on every screen change (rotation, window resize)

    -- the window's size at scale 1 (after the size limits) and the biggest scale at which it still fits the screen
    local baseW = math.clamp(MENU_SIZE.x * screenSize.X, MENU_MIN.X, MENU_MAX.X)
    local baseH = math.clamp(MENU_SIZE.y * screenSize.Y, MENU_MIN.Y, MENU_MAX.Y)
    local fit = math.min(screenSize.X * 0.96 / baseW, screenSize.Y * 0.96 / baseH)

    -- proportional scale based on how big the Roblox window currently is
    local propScale = math.min(screenSize.X / REF_SCREEN.X, screenSize.Y / REF_SCREEN.Y) * MENU_SCALE

    local targetScale = propScale
    local padding = 40

    -- phones / tablets have small screens, where the proportional scale would make the menu tiny:
    -- use the biggest scale (up to MENU_SCALE) at which the menu still fits the screen
    local kind = Nav.device.kind
    if (kind == "Phone" or kind == "Tablet") and targetScale < math.min(fit, MENU_SCALE) then
        targetScale = math.min(fit, MENU_SCALE)
    end

    -- still shrink further if the window gets too small for the menu's min size
    if screenSize.X < (MENU_MIN.X + padding) or screenSize.Y < (MENU_MIN.Y + padding) then
        local scaleX = screenSize.X / (MENU_MIN.X + padding)
        local scaleY = screenSize.Y / (MENU_MIN.Y + padding)
        targetScale = math.min(targetScale, math.min(scaleX, scaleY) * MENU_SCALE)
    end

    -- device profile (e.g. a console is viewed from far away, so it starts bigger); never past the screen edges
    local profile = Nav.profileScale[kind] or 1
    if profile ~= 1 then
        if Nav.fitToScreen == false then
            targetScale = targetScale * profile
        else
            targetScale = math.min(targetScale * profile, math.max(fit, targetScale))
        end
    end

    -- size multiplier chosen in Settings > Menu Size (1 = normal, up to 2)
    local boost = Nav.sizeBoost or 1
    Nav.normalScale = targetScale
    if boost ~= 1 then
        if Nav.fitToScreen == false then
            targetScale = targetScale * boost   -- no screen limit: the menu may grow past the screen edges
        else
            -- keep the menu inside the screen: never grow beyond 96% of the screen width / height
            targetScale = math.min(targetScale * boost, math.max(fit, targetScale))
        end
    end

    -- on big screens (4K, 65" TV) the menu grows proportionally to the resolution instead of stopping at MENU_SCALE
    -- (6 = upper scale limit; on 3840x2160 this gives roughly x1.9 at normal size)
    uiScale.Scale = math.clamp(targetScale, 0.45, 6)
    Nav.effectiveBoost = uiScale.Scale / math.max(Nav.normalScale, 0.001)   -- what was really applied (can be lower than asked)

    -- center the menu on screen using the new scale (used on startup and when opening via the "<" tab / RightShift)
    local winW = baseW * uiScale.Scale
    local winH = baseH * uiScale.Scale
    MENU_POS.x = math.clamp((screenSize.X - winW) / 2 / screenSize.X, 0, 1)
    MENU_POS.y = math.clamp((screenSize.Y - winH) / 2 / screenSize.Y, 0, 1)

    -- after a screen change (rotation, window resize) make sure an open menu is not left off-screen
    if Nav.keepOnScreen then task.delay(0.15, Nav.keepOnScreen) end
end

local function doSearch(term)
    clearHighlight()
    if term=="" then clearBtn.Visible=false; searchBox.PlaceholderText="Search elements..."; return end
    clearBtn.Visible=true
    term=string.lower(term)

    -- split the search term into individual words so "rainbow cursor"
    -- also matches an element named "Cursor Rainbow" (any word order)
    local queryWords = {}
    for w in term:gmatch("%S+") do table.insert(queryWords, w) end

    local found=0; local firstTab=nil
    for _,e in ipairs(searchElements) do
        local nameLower = string.lower(e.name)
        local matches
        if string.find(nameLower,term,1,true) then
            -- exact substring match (handles single-word queries / exact phrase)
            matches = true
        else
            -- otherwise require every query word to appear somewhere in the name
            matches = true
            for _,w in ipairs(queryWords) do
                if not string.find(nameLower,w,1,true) then
                    matches = false
                    break
                end
            end
        end
        if matches then
            found=found+1
            if e.elem and e.elem.Parent and e.elem:IsA("GuiObject") then
                local ot,oc=e.elem.BackgroundTransparency,e.elem.BackgroundColor3
                table.insert(searchHighlighted,{row=e.elem,t=ot,c=oc})
                e.elem.BackgroundColor3=scaledC(COLORS.accent,.18)
                e.elem.BackgroundTransparency=0
            end
            if not firstTab then firstTab=e.tab end
        end
    end
    searchBox.PlaceholderText=found>0 and found.." found" or "No results"
    if firstTab then
        local switchedSide = false
        for _,e in ipairs(Nav.topTabBtns) do
            if e.name==firstTab then selectTop(e.button, e.name); switchedSide = true; break end
        end
        if not switchedSide then
            for _,e in ipairs(Nav.sideButtons) do
                if e.name==firstTab then selectSide(e.button,e.gradient,e.name,e.page); switchedSide = true; break end
            end
        end
        if not switchedSide then
            for _,e in ipairs(Nav.rightTabBtns) do
                if e.name==firstTab then selectRight(e.button,e.name,Nav.rightTabPgs[e.name]); break end
            end
        end
    end
end

searchBox.Changed:Connect(function(p) if p=="Text" then doSearch(searchBox.Text) end end)

-- ===================== ACTION HISTORY (data only) =====================
-- Every recorded action goes into Nav.historyLog. The History TAB itself is built by the second module
-- (Window:AddRightTab) and draws the entries through Nav.historySink; the log is replayed to it when it connects,
-- so nothing is lost if that module loads a moment later.
Nav.historyLog = {}    -- oldest first: { text = string, color = Color3, revert = function or nil, time = "HH:MM:SS" }
Nav.historySink = nil  -- set with Dyser:SetHistorySink(fn)
addHistoryEntry = function(txt, col, revertFn)
    local entry = { text = txt, color = col, revert = revertFn, time = os.date and os.date("%H:%M:%S") or "" }
    table.insert(Nav.historyLog, entry)
    if #Nav.historyLog > 300 then table.remove(Nav.historyLog, 1) end   -- keep memory bounded
    if Nav.historySink then pcall(Nav.historySink, entry) end
end

    table.insert(scriptConnections,box.FocusLost:Connect(function() 
        if box.Text ~= prevText then
            local oldTxt = prevText; local newTxt = box.Text; prevText = newTxt
            if cb then cb(newTxt) end
            if addHistoryEntry then
                addHistoryEntry("⌨ " .. txt .. " → " .. newTxt, COLORS.text, function() box.Text = oldTxt; prevText = oldTxt; if cb then cb(oldTxt) end end)
            end
        end
    end))
    return box
end

-- Configurable keybind row: shows a "rebind" button (click then press a key)
-- and an ON/OFF toggle box. defaultKey is an Enum.KeyCode; onToggle(active)
-- fires whenever the bound key is pressed and the bind is enabled, or when
-- the toggle box itself is clicked directly.
local function addKeybind(txt, defaultKey, onToggle)
    local row=makeRow(23)
    if curSection then regSearch(txt,curSection.name,row) end
    rowLabel(row,txt, "Labels")

    local active=false
    local keyCode=defaultKey

    local toggleBox=label(row,"TextButton",{AnchorPoint=Vector2.new(1,.5),Position=UDim2.new(1,-3,.5,0), Size=UDim2.fromOffset(RS.n(SIZES.toggleBox),RS.n(SIZES.toggleBox)),BackgroundTransparency=0, BackgroundColor3=Color3.fromRGB(16,10,15),Text="x",TextColor3=COLORS.pink, TextSize=RS.n(18),Font=FONTS.bold,AutoButtonColor=false, Category="Buttons"})
    toggleBox.TextStrokeTransparency=.25; corner(toggleBox,4); bindCursorHover(toggleBox)
    local inactStr=stroke(toggleBox,Color3.fromRGB(48,28,43),1,.05)
    local glowStr=stroke(toggleBox,COLORS.accent,2,.0); glowStr.Transparency=1; table.insert(Acc.strokes,glowStr)

    local keyBtn=label(row,"TextButton",{AnchorPoint=Vector2.new(1,.5),Position=UDim2.new(1,-3-RS.n(SIZES.toggleBox)-6,.5,0), Size=UDim2.fromOffset(RS.n(64),RS.n(20)),BackgroundColor3=Color3.fromRGB(15,15,16), Text=keyCode and keyCode.Name or "None",TextColor3=COLORS.text,TextSize=RS.n(11),Font=Enum.Font.Code,AutoButtonColor=false, Category="Inputs"})
    corner(keyBtn,3); stroke(keyBtn,Color3.fromRGB(30,30,32),1,.2); bindCursorHover(keyBtn); trackText("Inputs", keyBtn); trackColorText(keyBtn)

    local entry = {keyCode=keyCode, button=keyBtn, active=false}

    local function setActive(v)
        local prevVal = active
        active=v; entry.active=v
        toggleBox.Text=v and "✓" or "x"
        toggleBox.TextColor3=v and Color3.fromRGB(70,230,120) or COLORS.pink
        toggleBox.BackgroundColor3=v and Color3.fromRGB(9,30,15) or Color3.fromRGB(16,10,15)
        row.BackgroundTransparency=v and .02 or .14
        TweenService:Create(glowStr,TweenInfo.new(.2),{Transparency=v and .1 or 1}):Play()
        inactStr.Transparency = v and 1 or .05
        if addHistoryEntry then
            playsfx(v and "toggleOn" or "toggleOff")
            addHistoryEntry((v and "✓ ON  " or "✗ OFF ")..txt, v and COLORS.accent or COLORS.pink, function() setActive(prevVal) end)
        end
        if onToggle then onToggle(v) end
    end
    entry.setActive = setActive

    row.MouseEnter:Connect(function() TweenService:Create(row,TweenInfo.new(.12),{BackgroundTransparency=0}):Play() end)
    row.MouseLeave:Connect(function() TweenService:Create(row,TweenInfo.new(.12),{BackgroundTransparency=active and .02 or .14}):Play() end)
    table.insert(scriptConnections,toggleBox.MouseButton1Click:Connect(function() setActive(not active) end))

    table.insert(scriptConnections,keyBtn.MouseButton1Click:Connect(function()
        playsfx("click")
        keyBtn.Text="..."
        waitingBind=entry
    end))

    table.insert(keybinds, entry)
    return entry
end

        local ry = math.clamp(input.Position.Y - hueSlider.AbsolutePosition.Y, 0, hueSlider.AbsoluteSize.Y)

    end)

        if #parts == 3 then
            local r = tonumber(parts[1]:match("%d+")); local g = tonumber(parts[2]:match("%d+")); local b = tonumber(parts[3]:match("%d+"))

        end
    end)

                if r and g and b then
                    local oldH, oldS, oldV = curH, curS, curV 
                    curH, curS, curV = Color3.fromRGB(r, g, b):ToHSV()
                    updateUI()
                    if addHistoryEntry then 
                        addHistoryEntry("🎨 Hex Changed: " .. txt, previewBtn.BackgroundColor3, function()
                            curH, curS, curV = oldH, oldS, oldV
                            updateUI()

                    end

        playsfx("click")
    end)

    updateUI()
end

-- Dropdown: a row that shows the current choice; click it to open a list under it, click an option to pick it.
-- opts = list of strings, startIdx = initially selected index, cb(index, text) is called when the choice changes.
local function addDropdown(txt, opts, startIdx, cb)
    local collapsedHeight = RS.n(28)
    local itemH = RS.n(24)
    local expandedHeight = collapsedHeight + #opts * itemH + 6
    local row = makeRow(28)
    row.ClipsDescendants = true

    local lbl = rowLabel(row, txt, "Labels")
    lbl.Size = UDim2.new(.4, 0, 0, collapsedHeight)
    lbl.TextYAlignment = Enum.TextYAlignment.Center
    if curSection then regSearch(txt, curSection.name, row) end

    local selected = math.clamp(startIdx or 1, 1, #opts)
    local expanded = false

    -- header button: shows the current choice
    local btn = label(row, "TextButton", {AnchorPoint=Vector2.new(1,0), Position=UDim2.new(1,-6,0,RS.n(3)),
        Size=UDim2.new(.56,0,0,collapsedHeight - RS.n(6)), BackgroundTransparency=0, BackgroundColor3=Color3.fromRGB(15,15,16),
        Text="", TextSize=RS.n(11), Font=Enum.Font.Code, AutoButtonColor=false, Category="Inputs"})
    corner(btn, 4); stroke(btn, COLORS.lineLight, 1, .2); bindCursorHover(btn)
    local btnText = label(btn, "TextLabel", {Position=UDim2.fromOffset(8,0), Size=UDim2.new(1,-RS.n(26),1,0),
        Text=opts[selected], TextSize=RS.n(11), Font=Enum.Font.Code, TextXAlignment=Enum.TextXAlignment.Left,
        TextTruncate=Enum.TextTruncate.AtEnd, Category="Inputs"})
    local arrow = label(btn, "TextLabel", {AnchorPoint=Vector2.new(1,.5), Position=UDim2.new(1,-6,.5,0),
        Size=UDim2.fromOffset(RS.n(14),RS.n(14)), Text="v", TextSize=RS.n(11), Font=FONTS.bold,
        TextColor3=COLORS.accent, TextXAlignment=Enum.TextXAlignment.Center, Category="Buttons"})
    table.insert(Acc.texts, arrow)

    -- option buttons live inside the row, under the header (the row is clipped, so they only show when it is expanded)
    local items = {}
    local function refresh()
        for i, it in ipairs(items) do
            local sel = (i == selected)
            it.Text = (sel and "✓ " or "   ") .. opts[i]
            it.BackgroundColor3 = sel and Color3.fromRGB(28,28,34) or Color3.fromRGB(12,12,14)
            it.Font = sel and FONTS.bold or Enum.Font.Code
        end
        btnText.Text = opts[selected]
    end
    local function setExpanded(v)
        expanded = v
        arrow.Text = v and "^" or "v"
        if v then for _, it in ipairs(items) do it.Visible = true end end
        TweenService:Create(row, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {Size = UDim2.new(1, 0, 0, v and expandedHeight or collapsedHeight)}):Play()
        if not v then
            task.delay(0.25, function()
                if not expanded then for _, it in ipairs(items) do it.Visible = false end end
            end)
        end
    end
    for i = 1, #opts do
        local it = label(row, "TextButton", {Position=UDim2.new(0,6,0,collapsedHeight + 2 + (i-1)*itemH),
            Size=UDim2.new(1,-12,0,itemH - 2), BackgroundTransparency=0, BackgroundColor3=Color3.fromRGB(12,12,14),
            Text="", TextSize=RS.n(11), Font=Enum.Font.Code, TextXAlignment=Enum.TextXAlignment.Left,
            AutoButtonColor=false, Category="Inputs"})
        it.Visible = false
        corner(it, 3); bindCursorHover(it)
        it.MouseButton1Click:Connect(function()
            selected = i
            refresh()
            playsfx("click")
            setExpanded(false)
            if cb then cb(i, opts[i]) end
        end)
        items[i] = it
    end
    refresh()
    btn.MouseButton1Click:Connect(function() playsfx("click"); setExpanded(not expanded) end)
end

-- ===================== BUILDER =====================
local function buildPage(page,rows,tabName)
    settingsPage=page; curSection=nil
    for _,r in ipairs(rows) do
        if     r[1]=="section"      then addSection(r[2],tabName)
        elseif r[1]=="toggle"       then addToggle(r[2],r[3])
        elseif r[1]=="button"       then addButton(r[2],r[3])
        elseif r[1]=="slider"       then addSlider(r[2],r[3],r[4],r[5])
        elseif r[1]=="input"        then local b=addInput(r[2],r[3],r[4]);
        elseif r[1]=="colorpicker"  then addColorPicker(r[2],r[3],r[4])
        elseif r[1]=="keybind"      then addKeybind(r[2],r[3],r[4])
        elseif r[1]=="rebind"       then addRebindOnly(r[2],r[3],r[4])
        elseif r[1]=="dropdown"     then addDropdown(r[2],r[3],r[4],r[5])
        end
    end
end

-- (любым способом — RightShift, кнопка "-", клик по кружку, мобильная кнопка)
-- не чаще одного раза в TOGGLE_COOLDOWN секунд
local TOGGLE_COOLDOWN = 1.0
local lastToggleTime = 0
local function canToggle()
    local now = os.clock()
    if now - lastToggleTime < TOGGLE_COOLDOWN then
        return false
    end
    lastToggleTime = now
    return true
end

-- круглая вспышка (кружок, minimize) — центр в точке, размер растёт как квадрат
local function pulseGlow(centerPos, startSize, endSize, duration)
    glowCorner.CornerRadius=UDim.new(0,1000)
    glowRing.Position=UDim2.fromOffset(centerPos.X, centerPos.Y)
    glowRing.Size=UDim2.fromOffset(startSize,startSize)
    glowRing.Visible=true
    glowStroke.Color=COLORS.accent
    glowStroke.Transparency=0.1
    glowStroke.Thickness=4
    TweenService:Create(glowRing,TweenInfo.new(duration,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),
        {Size=UDim2.fromOffset(endSize,endSize)}):Play()
    local fade=TweenService:Create(glowStroke,TweenInfo.new(duration,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),
        {Transparency=1, Thickness=0.5})
    fade:Play()
    fade.Completed:Connect(function()
        glowRing.Visible=false
    end)
end

-- Window glow (open / close / minimize / restore). It lives INSIDE the window in a clipped host frame, so it is
-- measured in the window's own units: it automatically follows every move / resize / UI scale of the window and can
-- never extend outside the menu frame (an overlay in the ScreenGui could, because its size had to be calculated by hand).
--   "open"  = the outline starts deep inside the window and ripples OUTWARD to the frame edge while fading
--   "close" = the outline starts at the frame edge and collapses INWARD while fading
Nav.wgHost = Instance.new("Frame", window)
Nav.wgHost.Name = "DyserWindowGlowHost"; Nav.wgHost.Size = UDim2.new(1,0,1,0)
Nav.wgHost.BackgroundTransparency = 1; Nav.wgHost.BorderSizePixel = 0
Nav.wgHost.ClipsDescendants = true; Nav.wgHost.Active = false; Nav.wgHost.ZIndex = 100
Nav.wgRing = Instance.new("Frame", Nav.wgHost)
Nav.wgRing.Name = "DyserWindowGlow"; Nav.wgRing.BackgroundTransparency = 1; Nav.wgRing.BorderSizePixel = 0
Nav.wgRing.Active = false; Nav.wgRing.ZIndex = 101; Nav.wgRing.Visible = false
Nav.wgRing.Size = UDim2.new(1,0,1,0)
corner(Nav.wgRing, 10)
Nav.wgStroke = Instance.new("UIStroke", Nav.wgRing)
Nav.wgStroke.Color = COLORS.accent; Nav.wgStroke.Thickness = 3; Nav.wgStroke.Transparency = 1
table.insert(Acc.strokes, Nav.wgStroke)   -- follows the menu accent color (including rainbow)

Nav.playGlow = function(kind, duration)
    Nav.glowToken = (Nav.glowToken or 0) + 1
    local token = Nav.glowToken
    local ring, stroke = Nav.wgRing, Nav.wgStroke
    duration = math.max(duration or .5, .05)
    stroke.Color = COLORS.accent
    ring.Visible = true
    local t0 = os.clock()
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if token ~= Nav.glowToken then conn:Disconnect(); return end   -- a newer effect took over
        local p = math.clamp((os.clock() - t0) / duration, 0, 1)
        local sz = window.AbsoluteSize / math.max(uiScale.Scale, 0.01)  -- window size in its own units
        local minDim = math.max(math.min(sz.X, sz.Y), 8)
        local depth = (kind == "open") and (1 - p) or p                 -- how far inside the window the outline sits
        local inset = math.min(3 + minDim*0.18*depth, minDim/2 - 2)
        ring.Position = UDim2.fromOffset(inset, inset)
        ring.Size = UDim2.new(1, -2*inset, 1, -2*inset)
        if kind == "open" then
            stroke.Transparency = 0.1 + 0.9*p*p
            stroke.Thickness = math.max(0.5, 5 - 4.5*p)
        else
            stroke.Transparency = 0.15 + 0.85*p
            stroke.Thickness = math.max(0.5, 4 - 3.5*p)
        end
        if p >= 1 then
            conn:Disconnect()
            ring.Visible = false
        end
    end)
end

-- open / restore / expand: the outline ripples outward (old arguments are ignored, only the duration is used)
local function pulseGlowRect(a1, a2, a3, a4, duration) Nav.playGlow("open", duration or .6) end

-- close / minimize: the outline collapses inward
local function pulseGlowShrink(a1, a2, a3, a4, duration) Nav.playGlow("close", duration or .5) end

-- AbsolutePosition and Position are NOT in the same coordinate space here: the menu's ScreenGui ignores the GUI inset, but
-- AbsolutePosition is measured from below Roblox's top bar, so  Position-space Y = AbsolutePosition.Y + inset  (the original
-- custom-cursor code relies on the same fact). Anything placed with Position (effects, the bubble) must therefore NOT be
-- computed from AbsolutePosition, otherwise it ends up one top-bar height too high.
Nav.guiInset = function()
    local ok, v = pcall(function() return game:GetService("GuiService"):GetGuiInset().Y end)
    return (ok and type(v) == "number") and v or 0
end
-- center of the collapsed bubble (its final size is 54x54), taken from its Position so it always matches where it is drawn
Nav.bubbleCenter = function()
    local pos, ss = collapseBubble.Position, screenGui.AbsoluteSize
    return Vector2.new(pos.X.Scale * ss.X + pos.X.Offset + 27, pos.Y.Scale * ss.Y + pos.Y.Offset + 27)
end

local function expandFromBubble()
    isCollapsed=false
    playsfx("open")

    -- зелёная вспышка-кольцо точно по центру кружка в момент его исчезновения
    pulseGlow(Nav.bubbleCenter(), 54, 130, .5)

    -- 1. пузырь медленно "сжимается" и тает, лого прокручивается напоследок
    local shrink = TweenService:Create(collapseBubble, TweenInfo.new(.32,Enum.EasingStyle.Quad,Enum.EasingDirection.In),
        {Size=UDim2.fromOffset(14,14), BackgroundTransparency=1})
    local fadeStroke = TweenService:Create(collapseStroke, TweenInfo.new(.32,Enum.EasingStyle.Quad,Enum.EasingDirection.In),
        {Transparency=1})
    local spinOut = TweenService:Create(collapseLogo, TweenInfo.new(.32,Enum.EasingStyle.Quad,Enum.EasingDirection.In),
        {Rotation=collapseLogo.Rotation+120, ImageTransparency=1})
    shrink:Play(); fadeStroke:Play(); spinOut:Play()
    shrink.Completed:Wait()

    -- зелёная вспышка размером с будущее окно, растёт из левого верхнего угла —
    -- такой же эффект, как при обычном открытии меню, а не круглый
    local restoreSize = isMin and UDim2.new((preCollapseSize or fullSize).X.Scale,(preCollapseSize or fullSize).X.Offset,0,30) or (preCollapseSize or fullSize)
    local finalPosForGlow = preCollapsePos or window.Position
    local glowW = restoreSize.X.Scale * screenGui.AbsoluteSize.X + restoreSize.X.Offset
    local glowH = restoreSize.Y.Scale * screenGui.AbsoluteSize.Y + restoreSize.Y.Offset
    local glowX = finalPosForGlow.X.Scale * screenGui.AbsoluteSize.X + finalPosForGlow.X.Offset
    local glowY = finalPosForGlow.Y.Scale * screenGui.AbsoluteSize.Y + finalPosForGlow.Y.Offset
    pulseGlowRect(Vector2.new(glowX, glowY), Vector2.new(0,0), glowW, glowH, .65)

    collapseBubble.Visible=false
    collapseLogo.Rotation=0
    collapseLogo.ImageTransparency=0
    collapseBubble.BackgroundTransparency=0
    collapseStroke.Transparency=.1

    -- 2. окно появляется из той же точки, маленьким и полностью прозрачным
    window.Visible=true
    window.Size=UDim2.fromOffset(40,40)
    window.Position=collapseBubble.Position
    window.BackgroundTransparency=1
    if windowStroke then windowStroke.Transparency=1 end
    topBar.BackgroundTransparency=1
    local savedBodyTransp = body.BackgroundTransparency
    body.BackgroundTransparency=1

    if not isMin then
        body.Visible=true
    end

    -- 3. растягиваем окно и одновременно проявляем его и контент внутри — медленно и мягко, без рывка
    TweenService:Create(window,TweenInfo.new(.75,Enum.EasingStyle.Quint,Enum.EasingDirection.Out),
        {Size=restoreSize, Position=preCollapsePos or window.Position, BackgroundTransparency=0}):Play()
    if windowStroke then
        TweenService:Create(windowStroke,TweenInfo.new(.65,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),
            {Transparency=.35}):Play()
    end
    -- контент проявляется чуть позже самого окна — лёгкая "задержка слоями"
    task.delay(.25, function()
        if topBar and topBar.Parent then
            TweenService:Create(topBar,TweenInfo.new(.45,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),{BackgroundTransparency=0}):Play()
        end
        if body and body.Parent then
            TweenService:Create(body,TweenInfo.new(.45,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),{BackgroundTransparency=savedBodyTransp}):Play()
        end
    end)
end

local function collapseToCircle()
    isCollapsed=true
    preCollapseSize=window.Size
    preCollapsePos=window.Position
    playsfx("click")

    -- 1. контент гаснет первым (медленный "выдох"), окно следом сжимается
    local savedBodyTransp = body.BackgroundTransparency
    TweenService:Create(topBar,TweenInfo.new(.3,Enum.EasingStyle.Quad,Enum.EasingDirection.In),{BackgroundTransparency=1}):Play()
    TweenService:Create(body,TweenInfo.new(.3,Enum.EasingStyle.Quad,Enum.EasingDirection.In),{BackgroundTransparency=1}):Play()
    task.wait(.1)

    -- заметный "вдох" перед сжатием для пружинистости
    local punch = TweenService:Create(window, TweenInfo.new(.15,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),
        {Size=UDim2.new(preCollapseSize.X.Scale,preCollapseSize.X.Offset+8,preCollapseSize.Y.Scale,preCollapseSize.Y.Offset+6)})
    punch:Play(); punch.Completed:Wait()

    local shrink=TweenService:Create(window,TweenInfo.new(.6,Enum.EasingStyle.Quint,Enum.EasingDirection.In),
        {Size=UDim2.fromOffset(40,40), BackgroundTransparency=1})
    if windowStroke then
        TweenService:Create(windowStroke,TweenInfo.new(.4,Enum.EasingStyle.Quad,Enum.EasingDirection.In),{Transparency=1}):Play()
    end
    shrink:Play()
    shrink.Completed:Wait()
    window.Visible=false
    window.BackgroundTransparency=0
    topBar.BackgroundTransparency=0
    body.BackgroundTransparency=savedBodyTransp
    if windowStroke then windowStroke.Transparency=.35 end

    -- 2. пузырь появляется на том же месте: маленький+прозрачный → растёт с лёгким overshoot, лого крутится при появлении
    do  -- same coordinate space as Position (see Nav.guiInset above), as pure offsets because the bubble drag code works with offsets
        local wp, ss = window.Position, screenGui.AbsoluteSize
        collapseBubble.Position=UDim2.fromOffset(wp.X.Scale*ss.X + wp.X.Offset, wp.Y.Scale*ss.Y + wp.Y.Offset)
    end
    collapseBubble.Size=UDim2.fromOffset(10,10)
    collapseBubble.BackgroundTransparency=1
    collapseBubble.Visible=true
    collapseStroke.Transparency=1
    collapseLogo.Rotation=-100
    collapseLogo.ImageTransparency=1

    -- зелёная вспышка-кольцо точно по центру того места, где появится кружок
    -- (финальный размер кружка 54x54, поэтому центр = позиция + 27)
    pulseGlow(Nav.bubbleCenter(), 54, 140, .6)

    TweenService:Create(collapseBubble,TweenInfo.new(.55,Enum.EasingStyle.Back,Enum.EasingDirection.Out),
        {Size=UDim2.fromOffset(54,54), BackgroundTransparency=0}):Play()
    TweenService:Create(collapseStroke,TweenInfo.new(.55,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),
        {Transparency=.1}):Play()
    TweenService:Create(collapseLogo,TweenInfo.new(.6,Enum.EasingStyle.Back,Enum.EasingDirection.Out),
        {Rotation=0, ImageTransparency=0}):Play()
end

resizeHint.MouseButton1Click:Connect(function()

end)

-- allow dragging the little bubble around the screen while collapsed
local bubbleDragging,bubbleDragStart,bubbleStartPos=false,nil,nil
local bubbleDidDrag=false
local BUBBLE_DRAG_THRESHOLD=4 -- pixels of movement before it counts as a drag, not a click

collapseBubble.MouseButton1Click:Connect(function()
    if isCollapsed and not bubbleDidDrag and canToggle() then expandFromBubble() end
end)

collapseBubble.InputBegan:Connect(function(i)
    if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
        bubbleDragging=true; bubbleDidDrag=false; bubbleDragStart=i.Position; bubbleStartPos=collapseBubble.Position
    end
end)
collapseBubble.InputEnded:Connect(function(i)
    if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
        bubbleDragging=false
    end
end)
table.insert(scriptConnections, UserInputService.InputChanged:Connect(function(i)
    if bubbleDragging and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
        local delta=i.Position-bubbleDragStart
        if not bubbleDidDrag and (math.abs(delta.X)>BUBBLE_DRAG_THRESHOLD or math.abs(delta.Y)>BUBBLE_DRAG_THRESHOLD) then
            bubbleDidDrag=true
        end
        collapseBubble.Position=UDim2.fromOffset(bubbleStartPos.X.Offset+delta.X, bubbleStartPos.Y.Offset+delta.Y)
    end
end))

-- ===================== DRAG =====================
-- The window is moved with closed-loop correction: every frame we compare where it SHOULD be (where it was when grabbed
-- + how far the mouse has moved since) with where it ACTUALLY is (AbsolutePosition) and nudge Position by the difference.
-- Only mouse DELTAS are used, so the result does not depend on the screen inset, the UI scale or the coordinate space,
-- and the window stays exactly under the cursor at the point where you grabbed it.
-- (all drag state lives in one table to save local variables: the main chunk is close to the 200-local limit)
local drag = { on = false, gain = 1 }
-- current pointer position: the finger that started the drag (touch) or the mouse
drag.point = function()
    if drag.touch then return Vector2.new(drag.touch.Position.X, drag.touch.Position.Y) end
    return UserInputService:GetMouseLocation()
end
table.insert(scriptConnections,topBar.InputBegan:Connect(function(i)
    local t = i.UserInputType
    if t==Enum.UserInputType.MouseButton1 or t==Enum.UserInputType.Touch then
        drag.on=true; drag.touch=(t==Enum.UserInputType.Touch) and i or nil
        drag.mouse0=drag.point(); drag.win0=window.AbsolutePosition
        drag.gain=1; drag.lastDelta=nil; drag.lastAbs=nil
    end
end))
table.insert(scriptConnections,UserInputService.InputEnded:Connect(function(i)
    local t = i.UserInputType
    if t==Enum.UserInputType.MouseButton1 or (t==Enum.UserInputType.Touch and i==drag.touch) then drag.on=false; drag.touch=nil end
end))
table.insert(scriptConnections,RunService.RenderStepped:Connect(function()
    if not drag.on or not drag.mouse0 then return end
    local screenSize = screenGui.AbsoluteSize
    local winSize = window.AbsoluteSize
    local want = drag.win0 + (drag.point() - drag.mouse0)
    -- where the window may go. Normally it must stay fully on screen. With Nav.allowOffscreen it may be pushed partly outside
    -- the screen (left / right / bottom), but a strip of Nav.offscreenMargin always stays visible and the top bar never goes
    -- above the top edge, so the window can always be grabbed back. It is never snapped if it started outside the range.
    local minX, maxX, minY, maxY
    if Nav.allowOffscreen then
        local keep = (Nav.offscreenMargin or 40) * uiScale.Scale
        minX, maxX = keep - winSize.X, screenSize.X - keep
        minY, maxY = 0, screenSize.Y - keep - Nav.guiInset()
    else
        minX, maxX = 0, math.max(0, screenSize.X - winSize.X)
        minY, maxY = 0, math.max(0, screenSize.Y - winSize.Y - Nav.guiInset())
    end
    want = Vector2.new(
        math.clamp(want.X, math.min(minX, drag.win0.X), math.max(maxX, drag.win0.X)),
        math.clamp(want.Y, math.min(minY, drag.win0.Y), math.max(maxY, drag.win0.Y)))
    local nowAbs = window.AbsolutePosition
    -- learn how many absolute pixels the window really moves per 1 unit of Position offset
    -- (exactly 1 normally; measured here so the drag stays correct even if the UI scale also scales the offset)
    if drag.lastDelta and drag.lastAbs and drag.lastDelta.Magnitude > 4 then
        local ratio = (nowAbs - drag.lastAbs).Magnitude / drag.lastDelta.Magnitude
        if ratio > 0.2 and ratio < 5 then drag.gain = ratio end
    end
    local err = want - nowAbs
    if err.Magnitude < 0.5 then drag.lastDelta = nil; return end
    local delta = err / drag.gain
    local pos = window.Position
    window.Position = UDim2.new(pos.X.Scale, pos.X.Offset + delta.X, pos.Y.Scale, pos.Y.Offset + delta.Y)
    drag.lastDelta, drag.lastAbs = delta, nowAbs
end))

-- ===================== TOGGLE KEY =====================
table.insert(scriptConnections,UserInputService.InputBegan:Connect(function(input,proc)
    if input.UserInputType==Enum.UserInputType.MouseButton1 and cur.inside then cursorPulse() end
    if input.UserInputType~=Enum.UserInputType.Keyboard then return end
    if input.KeyCode==Enum.KeyCode.Delete then
        if not screenGui or not screenGui.Parent then return end
        for _,c in ipairs(scriptConnections) do if c and c.Connected then c:Disconnect() end end
        table.clear(scriptConnections)
        local killTween=TweenService:Create(window,TweenInfo.new(.35,Enum.EasingStyle.Back,Enum.EasingDirection.In),
            {Size=UDim2.new(0,0,0,0),Position=UDim2.new(MENU_POS.x+MENU_SIZE.x/2,0,MENU_POS.y+MENU_SIZE.y/2,0)})
        UserInputService.MouseIconEnabled = true
        killTween:Play(); killTween.Completed:Wait()
        if screenGui then screenGui:Destroy() end
        return
    end
    if TOGGLE_KEY and input.KeyCode==TOGGLE_KEY then
        if not screenGui or not screenGui.Parent then return end
        if not canToggle() then return end
        if isCollapsed then
            -- меню свёрнуто в кружок: RightShift прячет/показывает именно кружок, круглым эффектом
            if collapseBubble.Visible then
                pulseGlow(Nav.bubbleCenter(), 54, 120, .5)
                local t=TweenService:Create(collapseBubble,TweenInfo.new(.3,Enum.EasingStyle.Quad,Enum.EasingDirection.In),
                    {Size=UDim2.fromOffset(10,10), BackgroundTransparency=1})
                TweenService:Create(collapseStroke,TweenInfo.new(.3,Enum.EasingStyle.Quad,Enum.EasingDirection.In),{Transparency=1}):Play()
                TweenService:Create(collapseLogo,TweenInfo.new(.3,Enum.EasingStyle.Quad,Enum.EasingDirection.In),{ImageTransparency=1}):Play()
                t:Play(); t.Completed:Wait()
                collapseBubble.Visible=false
            else
                collapseBubble.Visible=true
                collapseBubble.Size=UDim2.fromOffset(10,10)

-- ===================== MENU SIZE (used by the three buttons in Settings > Menu Size) =====================
-- mult is a multiplier on top of the automatic scale (1 = normal). The menu stays on screen and is re-centered.
Nav.setMenuSize = function(mult)
    if isMaxed then return end   -- a maximized window already fills the screen
    Nav.sizeBoost = mult
    autoScaleUI()   -- recalculates the scale and the centered position (MENU_POS) using the multiplier
    if Nav.refreshDevice then Nav.refreshDevice() end
    -- smoothly move the window to its new center (if the menu is minimized to a strip, keep its vertical position)
    local target = isMin and UDim2.new(MENU_POS.x,0,window.Position.Y.Scale,window.Position.Y.Offset)
        or UDim2.new(MENU_POS.x,0,MENU_POS.y,0)
    TweenService:Create(window,TweenInfo.new(.3,Enum.EasingStyle.Quint,Enum.EasingDirection.Out),{Position=target}):Play()
    -- if the screen is too small for the asked size, say what was really applied
    local eff = Nav.effectiveBoost or mult
    if math.abs(eff - mult) > 0.02 then
        AddLog(string.format("Menu size: x%s (limited by the screen to x%.2f)", tostring(mult), eff), COLORS.accent)
    else
        AddLog("Menu size: x"..tostring(mult), COLORS.accent)
    end
end

-- ===================== KEEP ON SCREEN (after rotation / window resize) =====================
Nav.keepOnScreen = function()
    if not screenGui or not screenGui.Parent or not window.Visible or isMaxed or isCollapsed then return end
    local screenSize = screenGui.AbsoluteSize
    local pos, size = window.AbsolutePosition, window.AbsoluteSize
    local x, y
    if Nav.allowOffscreen then
        -- off-screen dragging is allowed: only make sure a strip of the menu (and its top bar) is still reachable
        local keep = (Nav.offscreenMargin or 40) * uiScale.Scale
        x = math.clamp(pos.X, keep - size.X, math.max(keep - size.X, screenSize.X - keep))
        y = math.clamp(pos.Y, 0, math.max(0, screenSize.Y - keep - Nav.guiInset()))
    else
        x = math.clamp(pos.X, 0, math.max(0, screenSize.X - size.X))
        y = math.clamp(pos.Y, 0, math.max(0, screenSize.Y - size.Y - Nav.guiInset()))
    end
    if math.abs(x - pos.X) < 1 and math.abs(y - pos.Y) < 1 then return end
    local p = window.Position
    TweenService:Create(window, TweenInfo.new(.25, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
        {Position = UDim2.new(p.X.Scale, p.X.Offset + (x - pos.X), p.Y.Scale, p.Y.Offset + (y - pos.Y))}):Play()
end

-- ===================== REJOIN (Settings > Server > Rejoin) =====================
-- Rejoins the same server when other players are in it; if you are alone there, joins a fresh server of the same place.
Nav.rejoin = function()
    if Nav.rejoining then return end
    Nav.rejoining = true
    local Players = game:GetService("Players")
    local TeleportService = game:GetService("TeleportService")
    AddLog("Rejoining...", COLORS.accent)
    local ok, err = pcall(function()
        if #Players:GetPlayers() <= 1 then
            TeleportService:Teleport(game.PlaceId, Players.LocalPlayer)
        else
            TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, Players.LocalPlayer)
        end
    end)
    if not ok then
        AddLog("Rejoin failed: "..tostring(err), COLORS.pink)
        Nav.rejoining = false   -- allow another try (teleport can fail, e.g. inside Studio)
    else
        task.delay(10, function() Nav.rejoining = false end)   -- teleport started; unlock again if we are still here
    end
end

-- ── Вспомогательная функция: создаёт новую верхнюю вкладку с ОТДЕЛЬНОЙ боковой панелью
-- Каждая верхняя вкладка через API получает свою собственную боковую панель (изолированно
-- от системных вкладок Main/Profiles/Theme и друг от друга).

-- Переключает активную боковую вкладку внутри одной API-панели (data), с плавным
-- "свечением" (accent-обводка) у выбранной кнопки, чтобы было видно в какой вкладке ты сейчас
function selectApiSideTab(data, entry)
    for _, e in ipairs(data.sideButtons) do
        TweenService:Create(e.btn, TweenInfo.new(.15), {BackgroundColor3 = Color3.fromRGB(17,17,18)}):Play()
        if e.activeStroke then
            TweenService:Create(e.activeStroke, TweenInfo.new(.15), {Transparency = 1}):Play()
        end
        e.page.Visible = false
    end
    TweenService:Create(entry.btn, TweenInfo.new(.15), {BackgroundColor3 = COLORS.accent}):Play()
    if entry.activeStroke then
        TweenService:Create(entry.activeStroke, TweenInfo.new(.15), {Transparency = 0}):Play()
    end
    entry.page.Visible = true
    data.selectedSide = entry.page
    settingsPage = entry.page
end

-- ── SideTab объект — то, через что пользователь добавляет элементы
local function makeSideTabAPI(page, sideName)
    local api = {}
    local pageSection = nil -- текущая секция ИМЕННО ЭТОЙ страницы — сохраняется между вызовами,
                              -- чтобы toggle/button/slider попадали в rows той секции, что была
                              -- открыта последним AddSection() на этой же странице

    -- перед каждым вызовом builder-функции нужно переключить settingsPage на нужную страницу
    local function withPage(fn)
        local prevPage    = settingsPage
        local prevSection = curSection
        settingsPage = page
        curSection   = pageSection -- восстанавливаем секцию, в которую писали в прошлый раз
        fn()
        pageSection  = curSection  -- запоминаем (возможно обновлённую addSection'ом) секцию
        settingsPage = prevPage
        curSection   = prevSection
    end

    function api:AddSection(txt)
        withPage(function() addSection(txt, sideName) end)
        return self
    end

    function api:AddToggle(opts)
        -- opts: { Text, Default, Callback }
        withPage(function()
            local cb = opts.Callback
            addToggle(opts.Text or "Toggle", cb)
            -- применяем значение по умолчанию через toggleSetters
            if opts.Default and Nav.ipairts then end
            local key = opts.Text or "Toggle"
            if opts.Default ~= nil and toggleSet and toggleSet[key] then
                toggleSet[key](opts.Default)
            end
        end)
        return self
    end

    function api:AddButton(opts)
        -- opts: { Text, Callback, DoubleClick }
        withPage(function() addButton(opts.Text or "Button", opts.Callback) end)
        return self
    end

    function api:AddSlider(opts)
        -- opts: { Text, Min, Max, Default, Callback }
        local min  = opts.Min     or 0
        local max  = opts.Max     or 100
        local def  = opts.Default or min
        local scale = (max - min) > 0 and (def - min) / (max - min) or 0
        withPage(function() addSlider(opts.Text or "Slider", def, scale, opts.Callback) end)
        return self
    end

    function api:AddInput(opts)
        -- opts: { Text, Default, Callback }
        withPage(function() addInput(opts.Text or "Input", opts.Default or "", opts.Callback) end)
        return self
    end

    function api:AddKeybind(opts)
        -- opts: { Text, Default, Callback }
        withPage(function() addKeybind(opts.Text or "Keybind", opts.Default, opts.Callback) end)
        return self
    end

    function api:AddColorPicker(opts)
        -- opts: { Text, Default, Callback }
        withPage(function() addColorPicker(opts.Text or "Color", opts.Default or Color3.fromRGB(255,255,255), opts.Callback) end)
        return self
    end

    function api:AddDropdown(opts)
        -- opts: { Text, Options = { "a", "b", ... }, Default = index, Callback = function(index, text) }
        local options = opts.Options or {}
        if #options == 0 then return self end
        withPage(function() addDropdown(opts.Text or "Dropdown", options, opts.Default, opts.Callback) end)
        return self
    end

    function api:AddRebind(opts)
        -- opts: { Text, Default = Enum.KeyCode, Callback = function(keyCode) }
        -- a row with a key button: click it, then press the new key (used for the menu hotkey)
        withPage(function() addRebindOnly(opts.Text or "Rebind", opts.Default, opts.Callback) end)
        return self
    end

    -- доступ к самому ScrollingFrame если нужен для кастомного кода
    function api:GetPage() return page end

    return api
end

-- ── TopTab объект
local function makeTopTabAPI(topName)
    local api = {}

    function api:AddSideTab(sideName)
        local page = createSideTabEntry(topName, sideName)
        return makeSideTabAPI(page, sideName)
    end

    function api:GetName() return topName end

    return api
end
